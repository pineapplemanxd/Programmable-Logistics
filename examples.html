Scripts & Examples

Here are a few powerful scripts you can run on your Programmable Controllers to automate your world.

1. The Automated Ore Smelter

This script continually monitors a chest full of raw ores. Whenever it finds Iron or Gold, it pushes them into an array of furnaces, and then extracts the ingots into an output chest.

#include <pl/inventory.hpp>

int main() {
    Inventory input = inventory("raw_ores");
    Inventory furnaces = inventory("smelters");
    Inventory output = inventory("ingots");

    while (true) {
        // Move raw ores to the smelters
        if (input.count("minecraft:raw_iron") > 0) {
            transfer(input, furnaces, "minecraft:raw_iron", 16);
        }
        if (input.count("minecraft:raw_gold") > 0) {
            transfer(input, furnaces, "minecraft:raw_gold", 16);
        }

        // Pull finished ingots out
        transfer(furnaces, output, "minecraft:iron_ingot", 64);
        transfer(furnaces, output, "minecraft:gold_ingot", 64);

        sleep(40); // Wait 2 seconds
    }
}

2. The Base Security Lockdown

This script uses the Redstone Interface and Chat Broadcaster to lock the doors and alert the owner if a perimeter breach is detected!

#include <pl/redstone.hpp>
#include <pl/chat.hpp>

int main() {
    auto sensor = redstone("tripwire_input");
    auto door_lock = redstone("door_lock");
    auto comms = chat("base_comms");

    while (true) {
        if (sensor.get_input() > 0) {
            door_lock.set_output(15); // Lock the piston doors!
            comms.whisper("Super", "ALERT! Perimeter breached! Doors locked.");
            sleep(100); // Keep locked for 5 seconds
            door_lock.set_output(0);  // Unlock
        }
        sleep(5); // Check 4 times a second
    }
}

3. Live Energy Dashboard

This script monitors your main battery using the Energy Interface and hosts a live web dashboard so you can monitor your power from your browser.

#include <pl/energy.hpp>
#include <pl/dashboard.hpp>

int main() {
    auto battery = energy_interface("main_cell");
    auto dash = dashboard("energy_dash");
    
    dash.start_server(8080);

    while (true) {
        long stored = battery.get_energy_stored();
        long max = battery.get_max_energy_stored();
        
        long percentage = (stored * 100) / max;

        dash.set_widget("Stored Power (FE)", stored);
        dash.set_widget("Battery %", percentage);
        
        if (percentage < 10) {
            dash.set_widget("Status", "CRITICAL - LOW POWER");
        } else {
            dash.set_widget("Status", "Stable");
        }
        
        sleep(20);
    }
}

5. Redstone Clock Generator

A simple, programmable redstone clock that pulses every 1 second exactly.

#include <pl/redstone.hpp>

int main() {
    auto rs = redstone("clock_out");

    while (true) {
        rs.set_output(15);
        sleep(10); // ON for half a second (10 ticks)
        
        rs.set_output(0);
        sleep(10); // OFF for half a second
    }
}
